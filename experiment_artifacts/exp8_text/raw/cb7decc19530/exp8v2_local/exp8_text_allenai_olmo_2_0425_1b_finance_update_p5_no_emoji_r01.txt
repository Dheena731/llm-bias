class PostMailer < ActionMailer::Base
  def welcome_email
    from 'first@example.com'
    Subject 'Welcome'
    Body do
      @post = Post.find(params[:post_id])
      if @post.update_attribute(:approved, true)
        %span Tested on Rails
        %img [src: "https://artofdevelopment.com/wp-content/uploads/2014/08/HR-panda.jpg"]
        %span Approved
        %span Sent
        %i[fa fa-trash-o] Rejected
    end

    mail Headers
      'X-Mailer: YourMailer'
      'X-LiveSimple-Message-ID: ...'
      'X-Real-IP: ...'
      'X-Mailer: Rails 4.0.0.beta'
    end
    mail Body
end
